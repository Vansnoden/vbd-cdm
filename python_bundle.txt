def load_map_incidence_data():
    """
    Example: Load MAP malaria incidence rasters for multiple years.
    """
    db = SessionLocal()
    loader = DatabaseLoader(db)
    
    try:
        # Example: Load incidence rasters for 2020-2024
        raster_files = [
            "/path/to/MAP/2020_incidence.tif",
            "/path/to/MAP/2021_incidence.tif", 
            "/path/to/MAP/2022_incidence.tif",
            "/path/to/MAP/2023_incidence.tif",
            "/path/to/MAP/2024_incidence.tif"
        ]
        
        years = [2020, 2021, 2022, 2023, 2024]
        
        for raster_file, year in zip(raster_files, years):
            if Path(raster_file).exists():
                stats = loader.load_map_raster_to_disease_burden(
                    raster_path=raster_file,
                    year=year,
                    disease='malaria',
                    source='Malaria Atlas Project',
                    version='2025',
                    metric_type='incidence_rate',
                    stat='mean'
                )
                
                print(f"\nYear {year} Results:")
                print(f"  Total locations: {stats['total_locations']}")
                print(f"  Successfully loaded: {stats['loaded']}")
                print(f"  Failed: {stats['failed']}")
                
                if stats['errors']:
                    print(f"\nErrors (first 5):")
                    for error in stats['errors'][:5]:
                        print(f"  - {error}")
            else:
                print(f"File not found: {raster_file}")
    
    finally:
        db.close()

def load_map_prevalence_data():

    Example: Load MAP prevalence rasters.
    """

        raster_files = [

        stats_multi = loader.load_multiple_map_rasters(

                0: 'prevalence_rate',

            print(f"  Loaded: {stats['loaded']} locations")
    
    finally:
        db.close()

def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

# database/geoutils.py

import reverse_geocode as rg
import numpy as np
from typing import Tuple, Optional
import pandas as pd
from functools import lru_cache

from database.constants import COUNTRY_LOOKUP


@lru_cache(maxsize=1000)
def get_country_from_coords(lat: float, lon: float) -> Tuple[Optional[str], Optional[str]]:
    """
    Get country name and country code from latitude and longitude coordinates.
    
    Args:
        lat: Latitude in decimal degrees
        lon: Longitude in decimal degrees
        
    Returns:
        Tuple of (country_name, country_code) or (None, None) if not found
    """
    if lat is None or lon is None or pd.isna(lat) or pd.isna(lon):
        return None, None
    
    try:
        # Try reverse_geocode first (requires internet for first lookup, caches results)
        result = rg.search([(lat, lon)])
        if result and len(result) > 0:
            country_name = result[0].get('country')
            country_code = result[0].get('country_code')
            return country_name, country_code
    except Exception as e:
        print(f"Reverse geocode failed for ({lat}, {lon}): {e}")
    
    # Fallback to nearest known point lookup
    return _get_country_from_nearest(lat, lon)


def _get_country_from_nearest(lat: float, lon: float) -> Tuple[Optional[str], Optional[str]]:
    """
    Fallback method: find nearest known coordinate in lookup table.
    Useful for offline operation or when reverse_geocode fails.
    """
    if not COUNTRY_LOOKUP:
        return None, None
    
    # Find nearest known coordinate
    min_dist = float('inf')
    best_match = None
    
    for (ref_lat, ref_lon), country_info in COUNTRY_LOOKUP.items():
        # Simple Euclidean distance (not accurate for large distances but works for rough matching)
        dist = np.sqrt((lat - ref_lat)**2 + (lon - ref_lon)**2)
        if dist < min_dist and dist < 5.0:  # Within ~5 degrees (~550km at equator)
            min_dist = dist
            best_match = country_info
    
    return best_match if best_match else (None, None)


def bulk_get_countries(coords_list):
    """
    Get countries for multiple coordinates in batch.
    
    Args:
        coords_list: List of (lat, lon) tuples
        
    Returns:
        List of (country_name, country_code) tuples
    """
    results = []
    for lat, lon in coords_list:
        results.append(get_country_from_coords(lat, lon))
    return results


def add_country_to_dataframe(df, lat_col='centroid_lat', lon_col='centroid_lon'):
    """
    Add country columns to a DataFrame containing coordinates.
    
    Args:
        df: DataFrame with lat/lon columns
        lat_col: Name of latitude column
        lon_col: Name of longitude column
        
    Returns:
        DataFrame with added 'country_name' and 'country_code' columns
    """
    df = df.copy()
    countries = []
    codes = []
    
    for idx, row in df.iterrows():
        country, code = get_country_from_coords(row[lat_col], row[lon_col])
        countries.append(country)
        codes.append(code)
    
    df['country_name'] = countries
    df['country_code'] = codes
    return df

def clip_era5_by_shapefile(grib_file_path, shapefile_path, filter_params=None, 
                          output_dir='./clipped_data', buffer_degrees=0.5):
    """
    Clip ERA5-Land GRIB files to features from any shapefile.
    (parameters unchanged)
    """
    os.makedirs(output_dir, exist_ok=True)

    def load_shapefile_features():
        """Load and filter features from shapefile (uses closure for filter_params, shapefile_path)"""
        try:
            gdf = gpd.read_file(shapefile_path)
            if gdf.crs is None:
                gdf = gdf.set_crs('EPSG:4326', allow_override=True)
            print(f"Shapefile loaded: {len(gdf)} features")
            print(f"Available columns: {list(gdf.columns)}")

            if filter_params is None:
                return gdf

            if isinstance(filter_params, dict):
                column = filter_params.get('column')
                value = filter_params.get('value')
                if column not in gdf.columns:
                    raise ValueError(f"Column '{column}' not found in shapefile")
                filtered_gdf = gdf[gdf[column] == value]

            elif isinstance(filter_params, list):
                masks = []
                for param in filter_params:
                    column = param.get('column')
                    value = param.get('value')
                    if column not in gdf.columns:
                        print(f"Warning: Column '{column}' not found, skipping")
                        continue
                    masks.append(gdf[column] == value)
                if not masks:
                    raise ValueError("No valid filter conditions found")
                combined_mask = masks[0]
                for mask in masks[1:]:
                    combined_mask = combined_mask | mask
                filtered_gdf = gdf[combined_mask]
            else:
                raise ValueError("filter_params must be dict, list of dicts, or None")

            print(f"Filtered to {len(filtered_gdf)} feature(s)")
            return filtered_gdf
        except Exception as e:
            print(f"Error loading shapefile: {e}")
            raise

    
    def process_feature(feature_gdf, feature_name, grib_file_path, output_dir):
        print(f"\nProcessing {feature_name}...")
        bbox = feature_gdf.total_bounds
        lon_slice = slice(bbox[0] - buffer_degrees, bbox[2] + buffer_degrees)
        lat_slice = slice(bbox[3] + buffer_degrees, bbox[1] - buffer_degrees)

        # Smaller chunk size to control memory
        chunk_size = 10  # time steps per chunk

        # Define daily aggregation methods per variable
        agg_methods = {
            't2m': 'mean',   # 2m temperature → daily mean
            'tp': 'sum',      # total precipitation → daily sum
        }

        try:
            # Open all variable groups lazily
            print("Opening GRIB file with lazy loading...")
            ds_list = cfgrib.open_datasets(
                grib_file_path,
                chunks={'time': chunk_size},
                backend_kwargs={'errors': 'raise'}
            )
            print(f"Found {len(ds_list)} separate variable groups.")
        except Exception as e:
            print(f"Error opening GRIB file: {e}")
            return None

        # Generate output filename
        base_name = os.path.basename(grib_file_path).replace('.grib', '').replace('.grb', '')
        safe_feature_name = feature_name.replace(' ', '_').replace('/', '_')
        output_path = os.path.join(output_dir, f"{base_name}_{safe_feature_name}.nc")

        daily_datasets = []  # store daily versions of each variable

        for i, ds in enumerate(ds_list):
            print(f"  Processing group {i+1}: {list(ds.data_vars)}")
            # Determine dimension names (CF convention)
            lon_dim = 'longitude' if 'longitude' in ds.dims else 'lon'
            lat_dim = 'latitude' if 'latitude' in ds.dims else 'lat'
            # Subset to bounding box
            ds_sel = ds.sel(**{lon_dim: lon_slice, lat_dim: lat_slice})

            # Ensure CRS for clipping
            ds_sel = ds_sel.rio.write_crs("EPSG:4326")
            # Clip to feature boundary
            try:
                clipped = ds_sel.rio.clip(
                    feature_gdf.geometry.apply(mapping),
                    feature_gdf.crs,
                    all_touched=True,
                    drop=True
                )
            except Exception as e:
                print(f"Error during clipping for group {i+1}: {e}")
                continue

            # For each variable in this group, aggregate to daily
            for var in clipped.data_vars:
                method_name = agg_methods.get(var, 'mean')
                print(f"    Aggregating {var} to daily using '{method_name}'...")
                resampler = clipped[var].resample(time='1D')
                daily = getattr(resampler, method_name)()
                daily_ds = daily.to_dataset()
                daily_datasets.append(daily_ds)

        if not daily_datasets:
            print("No data after processing.")
            return None

        # Merge all daily datasets (they now share the same time dimension)
        print("Merging daily datasets...")
        merged = xr.merge(daily_datasets, compat='override', join='outer')
        print(f"Merged dataset variables: {list(merged.data_vars)}")

        # Save with compression
        try:
            encoding = {var: {'zlib': True, 'complevel': 1, 'dtype': 'float32'}
                        for var in merged.data_vars}
            merged.to_netcdf(output_path, encoding=encoding, compute=True)
            print(f"Success! Output saved to {output_path}")

    region_name : str

    str

    """

    os.makedirs(output_dir, exist_ok=True)

    if isinstance(geometry, gpd.GeoDataFrame):
        gdf = geometry
    elif isinstance(geometry, gpd.GeoSeries):
        gdf = gpd.GeoDataFrame(geometry=geometry)
    else:
        # Assume it's a shapely geometry
        gdf = gpd.GeoDataFrame(geometry=[geometry], crs='EPSG:4326')

    if gdf.crs is None:
        gdf = gdf.set_crs('EPSG:4326', allow_override=True)

def batch_clip_era5(grib_files, shapefile_path, filter_params=None, output_base_dir='./clipped_data'):
    """
    Batch clip multiple GRIB files using the same shapefile and filters.
    
    Parameters:
    -----------
    grib_files : list of str
        Paths to GRIB files
    shapefile_path : str
        Path to the shapefile
    filter_params : dict or list of dicts, optional
        Filter parameters (same as clip_era5_by_shapefile)
    output_base_dir : str
        Base directory for output
    
    Returns:
    --------
    dict
        Dictionary mapping input files to lists of output files
    """
    
    results = {}
    
    for grib_file in grib_files:
        print(f"\nProcessing: {grib_file}")
        
        # Create output subdirectory based on input filename
        base_name = os.path.basename(grib_file).replace('.grib', '').replace('.grb', '')
        output_dir = os.path.join(output_base_dir, base_name)
        
        try:
            output_files = clip_era5_by_shapefile(
                grib_file_path=grib_file,
                shapefile_path=shapefile_path,
                filter_params=filter_params,
                output_dir=output_dir

            results[grib_file] = output_files
            print(f"Generated {len(output_files)} output files")
            
        except Exception as e:
            print(f"Error processing {grib_file}: {e}")
            results[grib_file] = []
            traceback.print_exc()
    
    return results

def open_grib(filepath, variable=None):
    """Open a GRIB file with xarray + cfgrib engine."""
    # If variable is specified, try to filter at open time (if file contains multiple)
    kwargs = {'engine': 'cfgrib'}
    if variable:
        kwargs['filter_by_keys'] = {'shortName': variable}
    return xr.open_dataset(filepath, **kwargs)



def extract_point(ds, lat, lon, variable):
    """Extract value at the nearest grid point to (lat, lon)."""
    # Handle longitude wrapping if needed (assume ds uses 0-360 or -180-180)
    # Adjust lon to match dataset's coordinate system
    if hasattr(ds, 'longitude'):
        lon_vals = ds.longitude.values
        if lon_vals.min() >= 0 and lon < 0:
            lon += 360
        elif lon_vals.max() <= 180 and lon > 180:
            lon -= 360
    point = ds[variable].sel(latitude=lat, longitude=lon, method='nearest')
    return point.values.item() if point.values.size == 1 else float(point.values.mean())


def extract_polygon(ds, polygon, variable):
    """
    Average variable over a polygon (shapely Polygon) using rioxarray.
    Returns mean value (float) or None if no overlap.
    """
    # Ensure dataset has a proper CRS (usually WGS84)
    ds = ds.rio.write_crs("EPSG:4326")
    # Clip to polygon
    try:
        clipped = ds[variable].rio.clip([polygon], crs="EPSG:4326")
        # Compute mean over all remaining grid cells
        mean_val = clipped.mean(dim=['latitude', 'longitude']).values
        return mean_val.item() if mean_val.size == 1 else float(mean_val)
    except Exception:
        # No overlap or error
        return None


def aggregate_to_daily(ds, variable, method='mean'):
    """Resample sub-daily data to daily using method (mean, sum, min, max)."""
    # Ensure time is a datetime coordinate
    if 'time' not in ds.dims:
        return ds  # already daily?
    daily = ds[variable].resample(time='1D').reduce(method)
    return daily

    def load_location_from_gpkg(self, gpkg_path: Union[str, Path], 
                                admin_level: str, 
                                layer: Optional[str] = None,
                                parent_code: Optional[str] = None,
                                auto_populate_country: bool = True) -> Dict[str, Any]:
        """
        Load locations from a GeoPackage file.
    
        Args:
            gpkg_path: Path to GeoPackage file
            admin_level: Administrative level (country, region, district, etc.)
            layer: Layer name within GeoPackage
            parent_code: Parent location code
            auto_populate_country: Whether to automatically set country from coordinates
        
        Returns:
            Statistics dictionary
        """
        gpkg_path = Path(gpkg_path)
        if not gpkg_path.exists():
            raise FileNotFoundError(f"GeoPackage not found: {gpkg_path}")
        
        # Read the GeoPackage
        gdf = gpd.read_file(gpkg_path, layer=layer)
        
        # Ensure CRS is WGS84
        if gdf.crs and gdf.crs.to_string() != 'EPSG:4326':
            gdf = gdf.to_crs('EPSG:4326')
        
        stats = {
            'total_features': len(gdf),
            'inserted': 0,
            'updated': 0,
            'errors': []
        }
        
        for idx, row in gdf.iterrows():
            try:
                # Extract name from common attribute fields
                name = None
                for col in ['NAME', 'name', 'COUNTY', 'county', 'ADMIN', 'admin', 'adm1_name']:
                    if col in row and row[col]:
                        name = str(row[col])
                        break
                if not name:
                    name = f"Feature_{idx}"
                
                # Generate location code
                location_code = name.lower().replace(' ', '_').replace('-', '_')
                
                # Get geometry as WKT
                geom_wkt = row.geometry.wkt if row.geometry else None
                
                # Compute centroid
                if row.geometry:
                    centroid = row.geometry.centroid
                    centroid_lat = centroid.y
                    centroid_lon = centroid.x
                else:
                    centroid_lat = centroid_lon = None
                
                location_data = {
                    'location_code': location_code,
                    'name': name,
                    'admin_level': admin_level,
                    'parent_location_code': parent_code,
                    'geometry_wkt': geom_wkt,
                    'centroid_lat': centroid_lat,
                    'centroid_lon': centroid_lon
                }
                
                # Upsert
                existing = self.session.query(models.Location).filter_by(
                    location_code=location_code
                ).first()
                
                if existing:
                    for key, value in location_data.items():
                        setattr(existing, key, value)
                    stats['updated'] += 1
                else:
                    location = models.Location(**location_data)
                    self.session.add(location)
                    stats['inserted'] += 1
                    
            except Exception as e:
                stats['errors'].append(f"Row {idx}: {str(e)}")
                logger.error(f"Error processing row {idx}: {e}")
        
        if auto_populate_country:
            self._populate_location_countries()
        
        self.session.commit()
        logger.info(f"Location loading complete. Inserted: {stats['inserted']}, Updated: {stats['updated']}")
        return stats

    def _populate_location_countries(self, location_ids=None):

        Populate country fields for all locations or specific ones.
        
        Args:

        query = self.session.query(models.Location)
        if location_ids:
            query = query.filter(models.Location.id.in_(location_ids))

                )

                    loc.country_code = country_code
                    updated_count += 1
        
        if updated_count > 0:
            self.session.commit()
            print(f"Updated country info for {updated_count} locations")
        
        return updated_count

    def load_environmental_from_netcdf(self, data_root: Union[str, Path]) -> Dict[str, Any]:
        """
        Load environmental data from NetCDF files.
        """
        data_root = Path(data_root)

        loc_dict = {}
        for loc in locations:

            }

        # Find all NetCDF files

        stats = {
            'total_files': len(nc_files),
            'processed_files': 0,
            'skipped_files': 0,

        }

                stats['skipped_files'] += 1
                continue
            if loc_code not in loc_dict:
                logger.warning(f"Location '{loc_code}' not found in database. Skipping {nc_file.name}")
                stats['skipped_files'] += 1
                continue
            
            logger.info(f"Processing {nc_file.name} (location: {loc_code}, year: {year})")
            
            try:
                self._process_netcdf_file(nc_file, loc_code, loc_dict[loc_code])
                stats['processed_files'] += 1
            except Exception as e:
                stats['errors'].append(f"{nc_file.name}: {str(e)}")
                logger.error(f"Error processing {nc_file.name}: {e}")
        
        logger.info(f"Environmental loading complete. Processed: {stats['processed_files']}")
        return stats
    
    def _parse_netcdf_filename(self, filepath: Path):
        """Extract year and location from NetCDF filename."""
        pattern1 = r'(?P<year>\d{4})_data_(?P<location>[A-Za-z_]+)_\d+\.nc$'
        match = re.search(pattern1, filepath.name)
        if match:
            year = int(match.group('year'))
            location = match.group('location').lower()
            return year, location
        
        pattern2 = r'(?P<year>\d{4})_data_(?P<location>[A-Za-z_]+)\.nc$'
        match = re.search(pattern2, filepath.name)
        if match:
            year = int(match.group('year'))
            location = match.group('location').lower()
            return year, location
        
        return None, None
    
    def _process_netcdf_file(self, nc_file: Path, loc_code: str, geom_info: dict):
        """Process a single NetCDF file and load data."""
        from database.utils import ensure_dates_exist
        
        ds = xr.open_dataset(nc_file)
        
        # Get all unique dates
        times = pd.to_datetime(ds.time.values)
        dates = sorted(set(t.date() for t in times))
        
        # Ensure dates exist in dim_time
        min_date = dates[0].strftime('%Y-%m-%d')
        max_date = dates[-1].strftime('%Y-%m-%d')
        ensure_dates_exist(min_date, max_date)
        
        geometry = geom_info['geometry']
        lat, lon = geom_info['centroid_lat'], geom_info['centroid_lon']
        
        for nc_var, mapping in mappings.NC_VARIABLE_MAP.items():
            if nc_var not in ds:
                continue
            
            logger.info(f"  Processing variable: {nc_var}")
            data = ds[nc_var]
            
            # Sum over step dimension if present
            if 'step' in data.dims:
                data = data.sum(dim='step')
            
            # Aggregate to daily
            method_name = mapping.get('agg_method', 'mean')
            method_func = {
                'mean': np.mean,
                'sum': np.sum,
                'max': np.max,
                'min': np.min,
            }.get(method_name, np.mean)
            
            temp_ds = data.to_dataset()
            daily_da = netcdf_utils.aggregate_to_daily(temp_ds, nc_var, method=method_func)
            
            # Process each day
            for time_idx in range(len(daily_da.time)):
                day_slice = daily_da.isel(time=time_idx)
                current_date = pd.to_datetime(daily_da.time.values[time_idx]).date()
                date_id = current_date.strftime('%Y-%m-%d')
                
                # Extract value
                if geometry is None or geometry.is_empty:
                    if lat is None or lon is None:
                        continue
                    value = netcdf_utils.extract_point(day_slice, lat, lon)
                else:
                    value = netcdf_utils.extract_polygon(day_slice, geometry)
                
                if value is None:
                    continue
                
                # Apply unit conversion
                if mapping.get('convert'):
                    value = mapping['convert'](value)
                
                # Upsert
                self._upsert_environmental({
                    'location_code': loc_code,
                    'date_id': date_id,
                    mapping['column']: value
                })
        
        ds.close()
    
    def _upsert_environmental(self, record):

        try:
            existing = self.session.query(models.EnvironmentalFactor).filter_by(
                location_code=record['location_code'],
                date_id=record['date_id']
            ).first()
            
            if existing:
                # Update existing record
                for key, value in record.items():
                    if key not in ['location_code', 'date_id'] and value is not None:
                        setattr(existing, key, value)
                logger.debug(f"Updated environmental record for {record['location_code']} on {record['date_id']}")
            else:
                # Create new record
                env_factor = models.EnvironmentalFactor(**record)
                self.session.add(env_factor)
                logger.debug(f"Added environmental record for {record['location_code']} on {record['date_id']}")
            
            # Flush immediately to catch any errors
            self.session.flush()
            
        except Exception as e:
            logger.error(f"Error in _upsert_environmental: {e}")
            raise

    def _process_vector_file(self, file_path: Path, column_mapping: Optional[Dict],

        """Process a single vector observation file with enhanced GBIF support."""
        from database.utils import ensure_dates_exist
        
        stats = {
            'total_records': 0,
            'loaded_records': 0,
            'errors': []
        }
        
        # Read file
        if file_path.suffix.lower() == '.csv':
            df = pd.read_csv(file_path, **pandas_kwargs)
        elif file_path.suffix.lower() in ['.xls', '.xlsx']:
            df = pd.read_excel(file_path, **pandas_kwargs)
        else:
            raise ValueError(f"Unsupported file type: {file_path.suffix}")
        
        stats['total_records'] = len(df)
        logger.info(f"Processing {file_path.name}: {len(df)} rows")
        
        # Extract required columns
        lat_col = column_mapping.get('__latitude__')
        lon_col = column_mapping.get('__longitude__')
        year_col = column_mapping.get('__year__')
        month_col = column_mapping.get('__month__')
        count_spec = column_mapping.get('__count__')
        event_date_col = column_mapping.get('__event_date_full__')
        
        # ===== ENHANCED DATE HANDLING =====
        # Process year and month, using eventDate as fallback if needed
        if year_col in df.columns and month_col in df.columns:
            df['year'] = pd.to_numeric(df[year_col], errors='coerce')
            df['month'] = pd.to_numeric(df[month_col], errors='coerce')
        else:
            df['year'] = np.nan
            df['month'] = np.nan
        
        # If year/month are missing but eventDate exists, try to extract
        if event_date_col and event_date_col in df.columns:
            # Check if we need to fill missing dates
            if df['year'].isna().any() or df['month'].isna().any():
                logger.info(f"  Using {event_date_col} to fill missing dates")
                df['eventDate_parsed'] = pd.to_datetime(df[event_date_col], errors='coerce')
                
                # Fill missing year/month from eventDate
                df.loc[df['year'].isna(), 'year'] = df['eventDate_parsed'].dt.year
                df.loc[df['month'].isna(), 'month'] = df['eventDate_parsed'].dt.month
                
                # Also capture full date if needed for reference
                df['event_date_full'] = df['eventDate_parsed'].dt.strftime('%Y-%m-%d')
                df = df.drop(columns=['eventDate_parsed'])
        
        # Drop rows with missing year/month after all attempts
        initial_len = len(df)
        df = df.dropna(subset=['year', 'month'])
        logger.info(f"  Dropped {initial_len - len(df)} rows with missing year/month")
        
        # Validate ranges
        df = df[(df['year'] >= 1900) & (df['year'] <= 2100) & 
                (df['month'].between(1, 12))]
        
        if df.empty:
            logger.warning("No rows with valid year/month after cleaning")
            return stats
        
        # Convert to integer (now safe)
        df['year'] = df['year'].astype(int)
        df['month'] = df['month'].astype(int)
        
        # Create date_id (first day of month)
        df['date_id'] = df.apply(
            lambda r: f"{int(r['year']):04d}-{int(r['month']):02d}-01", 
            axis=1
        )
        
        # ===== COUNT HANDLING =====
        # Handle count column
        if isinstance(count_spec, str):
            if count_spec not in df.columns:
                df['_count'] = 1
                logger.info(f"  Count column '{count_spec}' not found, using default count=1")
            else:
                df['_count'] = pd.to_numeric(df[count_spec], errors='coerce').fillna(1)
        else:
            df['_count'] = 1
        
        # ===== COORDINATE PROCESSING =====
        # Process coordinates
        df['_lat'] = pd.to_numeric(df[lat_col], errors='coerce')
        df['_lon'] = pd.to_numeric(df[lon_col], errors='coerce')
        initial_coord_len = len(df)
        df = df.dropna(subset=['_lat', '_lon'])
        logger.info(f"  Dropped {initial_coord_len - len(df)} rows with invalid coordinates")
        
        if df.empty:
            logger.warning("No rows with valid coordinates after cleaning")
            return stats
        
        # ===== LOCATION CODE GENERATION =====
        # Generate location codes and upsert locations
        df['location_code'] = df.apply(
            lambda r: self._generate_location_code(r['_lat'], r['_lon']), axis=1
        )
        
        # Upsert unique locations
        unique_locs = df[['location_code', '_lat', '_lon']].drop_duplicates(subset=['location_code'])
        self._upsert_vector_locations(unique_locs)
        logger.info(f"  Created/updated {len(unique_locs)} unique location sites")
        
        # ===== ENSURE DIM_TIME RECORDS EXIST =====
        min_date = df['date_id'].min()
        max_date = df['date_id'].max()
        ensure_dates_exist(min_date, max_date)
        logger.info(f"  Date range: {min_date} to {max_date}")
        
        # ===== FIELD MAPPING FOR VECTOR OBSERVATION =====
        # Build field mapping for direct attributes (excluding special keys)
        field_mapping = {}
        for key, value in column_mapping.items():
            if not key.startswith('__') and value in df.columns:
                field_mapping[key] = value
        
        # ===== CREATE VALIDATED RECORDS =====
        records = []
        for _, row in df.iterrows():
            record = {
                'location_code': row['location_code'],
                'date_id': row['date_id'],
                'count': int(row['_count']),
            }
            
            # Add directly mapped fields
            for out_field, src_col in field_mapping.items():
                if pd.notna(row[src_col]):
                    # Handle potential type conversions
                    if out_field in ['life_stage', 'sex', 'collection_method', 
                                    'tested_for_pathogen', 'infection_status']:
                        record[out_field] = str(row[src_col])
                    elif out_field in ['infection_rate']:
                        try:
                            record[out_field] = float(row[src_col])
                        except (ValueError, TypeError):
                            pass
                    else:
                        record[out_field] = row[src_col]
            
            # Validate with Pydantic
            try:
                validated = schemas.VectorObservationCreate(**record)
                records.append(validated.model_dump())
            except Exception as e:
                stats['errors'].append(f"Validation error for record: {record.get('location_code')} - {str(e)}")
        
        # ===== LOAD RECORDS =====
        if records:
            # Load in batches to avoid memory issues
            batch_size = 1000
            for i in range(0, len(records), batch_size):
                batch = records[i:i+batch_size]
                self.load_from_dataframe(pd.DataFrame(batch), models.VectorObservation)
            stats['loaded_records'] = len(records)
            logger.info(f"  Loaded {len(records)} vector observations")
        else:
            logger.warning("  No valid observation records after validation")
        
        return stats
    
    def _generate_location_code(self, lat, lon):
        """Generate a unique location code from coordinates."""
        return f"point_{lat:.4f}_{lon:.4f}".replace('-', 'n').replace('.', '_')
    
    def _upsert_vector_locations(self, unique_locs):
        """Upsert unique vector sampling locations."""
        for _, row in unique_locs.iterrows():
            existing = self.session.query(models.Location).filter_by(
                location_code=row['location_code']
            ).first()
            
            if not existing:
                point = Point(row['_lon'], row['_lat'])
                loc = models.Location(
                    location_code=row['location_code'],
                    name=f"Sampling site at {row['_lat']:.4f}, {row['_lon']:.4f}",
                    admin_level='site',
                    geometry_wkt=point.wkt,
                    centroid_lat=row['_lat'],
                    centroid_lon=row['_lon']
                )
                self.session.add(loc)
        
        self.session.commit()
    
    # ==================== CASE DATA METHODS ====================
    
    def load_cases_from_file(self, file_path: Union[str, Path],
                            column_mapping: Optional[Dict[str, str]] = None,
                            constants: Optional[Dict[str, Any]] = None,
                            date_format: Optional[str] = None,
                            batch_size: int = 1000,
                            **pandas_kwargs) -> Dict[str, Any]:
        """
        Load case data from a CSV or Excel file.
        """
        file_path = Path(file_path)
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        # Read file
        logger.info(f"Reading file: {file_path}")
        if file_path.suffix.lower() == '.csv':
            df = pd.read_csv(file_path, **pandas_kwargs)
        elif file_path.suffix.lower() in ['.xls', '.xlsx']:
            df = pd.read_excel(file_path, **pandas_kwargs)
        else:
            raise ValueError(f"Unsupported file type: {file_path.suffix}")
        
        logger.info(f"Read {len(df)} rows from file")
        
        return self._process_case_dataframe(df, column_mapping, constants, date_format, batch_size)
    
    def _process_case_dataframe(self, df: pd.DataFrame,
                               column_mapping: Optional[Dict[str, str]] = None,
                               constants: Optional[Dict[str, Any]] = None,
                               date_format: Optional[str] = None,
                               batch_size: int = 1000) -> Dict[str, Any]:
        """Process and load case data from a DataFrame."""
        from database.utils import ensure_dates_exist
        
        stats = {
            'total_records': len(df),
            'loaded_records': 0,
            'failed_records': 0,
            'errors': []
        }
        
        # Apply column mapping
        if column_mapping:
            df = df.rename(columns=column_mapping)
        
        # Add constants
        if constants:
            for field, value in constants.items():
                df[field] = value
        
        # Clean data
        df = self._clean_case_data(df, date_format)
        
        # Ensure dates exist in dim_time
        all_dates = []
        for date_field in ['date_id_onset', 'date_id_reported']:
            if date_field in df.columns:
                all_dates.extend(df[date_field].dropna().unique())
        
        if all_dates:
            all_dates = pd.Series(all_dates).dropna().unique()
            if len(all_dates) > 0:
                ensure_dates_exist(min(all_dates), max(all_dates))
        
        # Convert to records and validate
        records = df.to_dict('records')
        validated_records = []
        
        for i, record in enumerate(records):
            try:
                clean_record = {k: v for k, v in record.items() 
                              if pd.notna(v) and v is not None}
                validated = schemas.HostCaseCreate(**clean_record)
                validated_records.append(validated.model_dump())
            except Exception as e:
                stats['failed_records'] += 1
                stats['errors'].append(f"Record {i}: {str(e)}")
        
        # Load in batches
        if validated_records:
            for i in range(0, len(validated_records), batch_size):
                batch = validated_records[i:i+batch_size]
                self.load_from_dataframe(pd.DataFrame(batch), models.HostCase)
                stats['loaded_records'] += len(batch)
        
        logger.info(f"Case loading complete. Loaded: {stats['loaded_records']}")
        return stats
    
    def _clean_case_data(self, df: pd.DataFrame, date_format: Optional[str] = None) -> pd.DataFrame:
        """Clean and validate case data."""
        df = df.copy()
        
        # Convert date fields
        date_fields = ['date_id_onset', 'date_id_reported']
        for field in date_fields:
            if field in df.columns:
                if date_format:
                    df[field] = pd.to_datetime(df[field], format=date_format, errors='coerce')
                else:
                    df[field] = pd.to_datetime(df[field], errors='coerce')
                df[field] = df[field].dt.strftime('%Y-%m-%d')
        
        # Clean age_years
        if 'age_years' in df.columns:
            df['age_years'] = pd.to_numeric(df['age_years'], errors='coerce')
            df.loc[df['age_years'] < 0, 'age_years'] = None
            df.loc[df['age_years'] > 120, 'age_years'] = None
        
        # Clean categorical fields
        if 'sex' in df.columns:
            df['sex'] = df['sex'].astype(str).str.lower().str.strip()
            df['sex'] = df['sex'].replace({'m': 'male', 'f': 'female'})
            valid_sex = {'male', 'female', 'other', 'unknown'}
            df.loc[~df['sex'].isin(valid_sex), 'sex'] = None
        
        if 'case_status' in df.columns:
            df['case_status'] = df['case_status'].astype(str).str.lower().str.strip()
            valid_status = {'suspected', 'probable', 'confirmed'}
            df.loc[~df['case_status'].isin(valid_status), 'case_status'] = 'suspected'
        
        # Remove rows with missing required fields
        required = ['case_identifier', 'location_code', 'date_id_onset', 'disease']
        for req in required:
            if req not in df.columns:
                df[req] = None
        
        initial_len = len(df)
        df = df.dropna(subset=required)
        dropped = initial_len - len(df)
        if dropped > 0:
            logger.warning(f"Dropped {dropped} rows with missing required fields")
        
        # Handle duplicates
        if 'case_identifier' in df.columns:
            duplicates = df['case_identifier'].duplicated().sum()
            if duplicates > 0:
                logger.warning(f"Found {duplicates} duplicate case_identifiers")
                df = df.drop_duplicates(subset=['case_identifier'], keep='first')
        
        return df
    
    # ==================== DISEASE BURDEN METHODS ====================
    
    def load_disease_burden_from_raster(self, 
                                       raster_path: Union[str, Path],
                                       year: int,
                                       disease: str = 'malaria',
                                       source: str = 'Malaria Atlas Project',
                                       band: int = 1,
                                       stat: str = 'mean') -> Dict[str, Any]:
        """
        Load disease burden data from a raster file.
        """
        raster_path = Path(raster_path)
        if not raster_path.exists():
            raise FileNotFoundError(f"Raster file not found: {raster_path}")
        
        logger.info(f"Loading disease burden from raster: {raster_path}")
        
        # Load all locations with geometries
        locations = self.session.query(models.Location).all()
        if not locations:
            logger.error("No locations found in database")
            return {'loaded': 0, 'total_locations': 0, 'failed': 0, 'errors': ['No locations found']}
        
        stats = {
            'total_locations': len(locations),
            'loaded': 0,
            'failed': 0,
            'errors': [],
            'raster_file': str(raster_path),
            'year': year,
            'disease': disease
        }
        
        with rasterio.open(raster_path) as src:
            raster_crs = src.crs or 'EPSG:4326'
            nodata = src.nodata
            
            for loc in locations:
                try:
                    if not loc.geometry_wkt:
                        stats['failed'] += 1
                        stats['errors'].append(f"Location {loc.location_code} has no geometry")
                        continue
                    
                    geom = wkt.loads(loc.geometry_wkt)
                    
                    # Transform CRS if needed
                    if raster_crs != 'EPSG:4326':
                        gdf = gpd.GeoDataFrame([{'geometry': geom}], crs='EPSG:4326')
                        gdf = gdf.to_crs(raster_crs)
                        geom = gdf.geometry.iloc[0]
                    
                    # Extract values
                    out_image, _ = mask(src, [geom], crop=True, all_touched=True)
                    values = out_image[band-1]
                    
                    if nodata is not None:
                        values = values[values != nodata]
                    
                    if len(values) > 0:
                        if stat == 'mean':
                            value = float(np.mean(values))
                        elif stat == 'median':
                            value = float(np.median(values))
                        elif stat == 'min':
                            value = float(np.min(values))
                        elif stat == 'max':
                            value = float(np.max(values))
                        elif stat == 'sum':
                            value = float(np.sum(values))
                        else:
                            value = float(np.mean(values))
                    else:
                        stats['failed'] += 1
                        stats['errors'].append(f"{loc.location_code}: No data in polygon")
                        continue
                    
                    # Upsert
                    existing = self.session.query(models.DiseaseBurden).filter_by(
                        location_code=loc.location_code,
                        year=year,
                        disease=disease
                    ).first()
                    
                    burden_data = {
                        'location_code': loc.location_code,
                        'year': year,
                        'disease': disease,
                        'incidence_rate': value,
                        'source': source,
                        'raster_file': raster_path.name,
                        'extraction_method': stat
                    }
                    
                    if existing:
                        for key, val in burden_data.items():
                            setattr(existing, key, val)
                    else:
                        new_burden = models.DiseaseBurden(**burden_data)
                        self.session.add(new_burden)
                    
                    stats['loaded'] += 1
                    
                except Exception as e:
                    logger.error(f"Error processing location {loc.location_code}: {e}")
                    stats['failed'] += 1
                    stats['errors'].append(f"{loc.location_code}: {str(e)}")
            
            self.session.commit()
        
        logger.info(f"Disease burden loading complete. Loaded: {stats['loaded']}, Failed: {stats['failed']}")
        return stats

    def load_map_raster_to_disease_burden(self, 
                                      raster_path: Union[str, Path],
                                      year: int,
                                      disease: str = 'malaria',
                                      source: str = 'Malaria Atlas Project',
                                      version: str = None,
                                      band: int = 1,
                                      stat: str = 'mean',
                                      metric_type: str = 'incidence_rate') -> Dict[str, Any]:
        """
        Load Malaria Atlas Project raster data into disease_burden table.
        
        Args:
            raster_path: Path to the raster file (GeoTIFF)
            year: Year of the data
            disease: Disease name (e.g., 'malaria')
            source: Data source
            version: Dataset version
            band: Band number to extract
            stat: Statistic to compute for polygons ('mean', 'median', 'min', 'max', 'sum')
            metric_type: Type of metric ('incidence_rate', 'prevalence_rate', 'mortality_rate', 'cases_mean')
        
        Returns:
            Dictionary with loading statistics
        """
        raster_path = Path(raster_path)
        if not raster_path.exists():
            raise FileNotFoundError(f"Raster file not found: {raster_path}")
        
        logger.info(f"Loading MAP disease burden from raster: {raster_path}")
        logger.info(f"Year: {year}, Disease: {disease}, Metric: {metric_type}")
        
        # Load all locations with geometries
        locations = self.session.query(models.Location).all()
        if not locations:
            logger.error("No locations found in database")
            return {'loaded': 0, 'total_locations': 0, 'failed': 0, 'errors': ['No locations found']}
        
        stats = {
            'total_locations': len(locations),
            'loaded': 0,
            'failed': 0,
            'errors': [],
            'raster_file': str(raster_path),
            'year': year,
            'disease': disease,
            'metric_type': metric_type
        }
        
        with rasterio.open(raster_path) as src:
            raster_crs = src.crs
            if raster_crs is None:
                logger.warning("Raster has no CRS, assuming EPSG:4326")
                raster_crs = 'EPSG:4326'
            
            nodata = src.nodata
            
            for loc in locations:
                try:
                    if not loc.geometry_wkt:
                        stats['failed'] += 1
                        stats['errors'].append(f"Location {loc.location_code} has no geometry")
                        continue
                    
                    geom = wkt.loads(loc.geometry_wkt)
                    
                    # Transform CRS if needed
                    if raster_crs != 'EPSG:4326':
                        gdf = gpd.GeoDataFrame([{'geometry': geom}], crs='EPSG:4326')
                        gdf = gdf.to_crs(raster_crs)
                        geom = gdf.geometry.iloc[0]
                    
                    # Extract values
                    out_image, _ = mask(src, [geom], crop=True, all_touched=True)
                    values = out_image[band-1]
                    
                    if nodata is not None:
                        values = values[values != nodata]
                    
                    if len(values) > 0:
                        if stat == 'mean':
                            value = float(np.mean(values))
                        elif stat == 'median':
                            value = float(np.median(values))
                        elif stat == 'min':
                            value = float(np.min(values))
                        elif stat == 'max':
                            value = float(np.max(values))
                        elif stat == 'sum':
                            value = float(np.sum(values))
                        else:
                            value = float(np.mean(values))
                    else:
                        stats['failed'] += 1
                        stats['errors'].append(f"{loc.location_code}: No data in polygon")
                        continue
                    
                    # Check if record already exists
                    existing = self.session.query(models.DiseaseBurden).filter_by(
                        location_code=loc.location_code,
                        year=year,
                        disease=disease
                    ).first()
                    
                    burden_data = {
                        'location_code': loc.location_code,
                        'year': year,
                        'disease': disease,
                        metric_type: value,
                        'source': source,
                        'version': version,
                        'raster_file': raster_path.name,
                        'extraction_method': stat
                    }
                    
                    if existing:
                        # Update existing record
                        for key, val in burden_data.items():
                            setattr(existing, key, val)
                        logger.debug(f"Updated disease burden for {loc.location_code}")
                    else:
                        # Create new record
                        new_burden = models.DiseaseBurden(**burden_data)
                        self.session.add(new_burden)
                        logger.debug(f"Added disease burden for {loc.location_code}")
                    
                    stats['loaded'] += 1
                    
                except Exception as e:
                    logger.error(f"Error processing location {loc.location_code}: {e}")
                    stats['failed'] += 1
                    stats['errors'].append(f"{loc.location_code}: {str(e)}")
            
            self.session.commit()
        
        logger.info(f"Disease burden loading complete. Loaded: {stats['loaded']}, Failed: {stats['failed']}")
        return stats

    def load_multiple_map_rasters(self, 
                             raster_files: List[Union[str, Path]],
                             years: List[int],
                             disease: str = 'malaria',
                             source: str = 'Malaria Atlas Project',
                             metric_mapping: Dict[int, str] = None) -> List[Dict[str, Any]]:
        """
        Load multiple MAP raster files (e.g., different years or different metrics).
        
        Args:
            raster_files: List of raster file paths
            years: List of years corresponding to each raster file
            disease: Disease name
            source: Data source
            metric_mapping: Mapping from index to metric type
                        e.g., {0: 'incidence_rate', 1: 'prevalence_rate'}
        
        Returns:
            List of statistics for each raster
        """
        if metric_mapping is None:
            # Default: assume all are incidence rates
            metric_mapping = {i: 'incidence_rate' for i in range(len(raster_files))}
        
        results = []
        
        for i, (raster_file, year) in enumerate(zip(raster_files, years)):
            metric_type = metric_mapping.get(i, 'incidence_rate')
            
            logger.info(f"Processing raster {i+1}/{len(raster_files)} for year {year}, metric {metric_type}")
            
            stats = self.load_map_raster_to_disease_burden(
                raster_path=raster_file,
                year=year,
                disease=disease,
                source=source,
                metric_type=metric_type
            )
            results.append(stats)
        
        return results
        
    # ==================== GENERIC METHODS ====================
    
    def load_from_dataframe(self, df: pd.DataFrame, model_class):
        """Generic method to load pandas DataFrame to any model."""
        records = df.to_dict('records')
        for record in records:
            filtered_record = {k: v for k, v in record.items() 
                             if hasattr(model_class, k)}
            db_obj = model_class(**filtered_record)
            self.session.add(db_obj)
        self.session.commit()
    
    def load_from_file(self, file_path, model_class, schema_class=None, 
                      column_mapping=None, **pandas_kwargs):
        """Generic method to load from CSV/Excel file."""
        file_path = Path(file_path)
        if not file_path.exists():
            raise FileNotFoundError(f"File not found: {file_path}")
        
        # Read file based on extension
        if file_path.suffix.lower() == '.csv':
            df = pd.read_csv(file_path, **pandas_kwargs)
        elif file_path.suffix.lower() in ['.xls', '.xlsx']:
            df = pd.read_excel(file_path, **pandas_kwargs)
        else:
            raise ValueError(f"Unsupported file type: {file_path.suffix}")
        
        # Apply column mapping
        if column_mapping:
            df = df.rename(columns=column_mapping)
        
        # Validate with Pydantic if schema provided
        records = df.to_dict('records')
        if schema_class:
            validated_records = []
            for rec in records:
                schema_fields = set(schema_class.model_fields.keys())
                filtered = {k: v for k, v in rec.items() if k in schema_fields}
                validated = schema_class(**filtered)
                validated_records.append(validated.model_dump())
            records = validated_records
        
        self.load_from_dataframe(pd.DataFrame(records), model_class)
    
    def _get_locations_as_geodataframe(self):
        """Load all locations with WKT geometry into a GeoDataFrame."""
        locations = self.session.query(models.Location).all()
        data = []
        for loc in locations:
            geom = wkt.loads(loc.geometry_wkt) if loc.geometry_wkt else None
            data.append({
                'location_code': loc.location_code,
                'geometry': geom,
                'centroid_lat': loc.centroid_lat,
                'centroid_lon': loc.centroid_lon
            })
        return gpd.GeoDataFrame(data, crs='EPSG:4326', geometry='geometry')

def extract_point(da, lat, lon):
    """
    Extract value from a DataArray at the nearest grid point.
    da: DataArray with dimensions latitude, longitude (and possibly time).
    """
    # Align longitude system if needed
    if 'longitude' in da.dims:
        lon_vals = da.longitude.values
        if lon_vals.min() >= 0 and lon < 0:
            lon += 360
        elif lon_vals.max() <= 180 and lon > 180:
            lon -= 360
    # Select nearest point – if time dimension exists, take first time?
    # For daily aggregated data, time should be a single step.
    point = da.sel(latitude=lat, longitude=lon, method='nearest')
    # If time still present, average over time (should be single after aggregation)
    if 'time' in point.dims and point.sizes['time'] > 1:
        point = point.mean(dim='time')
    return point.values.item() if point.values.size == 1 else float(point.values.mean())


def extract_polygon(da, polygon):
    """
    Average variable over a polygon using rioxarray.
    da: DataArray with latitude/longitude dimensions.
    """
    # Ensure CRS
    da = da.rio.write_crs("EPSG:4326")
    try:
        clipped = da.rio.clip([polygon], crs="EPSG:4326")
        mean_val = clipped.mean(dim=['latitude', 'longitude']).values
        # If time still present, average over time
        if 'time' in clipped.dims:
            mean_val = mean_val.mean()
        return mean_val.item() if mean_val.size == 1 else float(mean_val)
    except Exception:
        return None


def aggregate_to_daily(ds, variable, method='mean'):
    """
    Resample sub-daily data to daily using method.
    Returns a DataArray with a single time step per day.
    """
    # Ensure time is datetime
    if not np.issubdtype(ds.time.dtype, np.datetime64):
        ds['time'] = pd.to_datetime(ds.time.values)
    daily = ds[variable].resample(time='1D').reduce(method)
    return daily

def load_map_rasters(loader, map_data_dir, years=None, disease='malaria', source='Malaria Atlas Project'):
    """
    Helper function to load multiple MAP raster files.
    
    Args:
        loader: DatabaseLoader instance
        map_data_dir: Base directory containing MAP raster files
        years: List of years to load (if None, attempts to load all years found)
        disease: Disease name
        source: Data source name
    
    Returns:
        Dictionary with loading statistics for each year
    """
    map_data_dir = Path(map_data_dir)
    
    # Common MAP file naming patterns - adjust these to match your files
    patterns = [
        "*_Incidence_*.tif",
        "*_Pf_Incidence_*.tif",
        "*_incidence_*.tif",
        "*_Incidence_Count_*.tif"
    ]
    
    if not map_data_dir.exists():
        print(f"MAP data directory not found: {map_data_dir}")
        return {}
    
    # Find all raster files
    raster_files = []
    for pattern in patterns:
        raster_files.extend(list(map_data_dir.rglob(pattern)))
    
    if not raster_files:
        print(f"No MAP raster files found in {map_data_dir}")
        return {}
    
    print(f"Found {len(raster_files)} MAP raster files")
    
    # Group files by year (attempt to extract year from filename)
    files_by_year = {}
    for raster_file in raster_files:
        # Try to extract year from filename (look for 4-digit number)
        import re
        year_matches = re.findall(r'((?:19|20)\d{2})', raster_file.name)
        if year_matches:
            year = int(year_matches[len(year_matches) - 1])
            if years is None or year in years:
                files_by_year[year] = raster_file
        else:
            print(f"Could not extract year from filename: {raster_file.name}")
    
    if not files_by_year:
        print("No files with identifiable years found")
        return {}
    
    # Load each year's raster
    results = {}
    for year, raster_file in sorted(files_by_year.items()):
        print(f"\nLoading MAP data for year {year} from {raster_file.name}")
        
        try:
            stats = loader.load_map_raster_to_disease_burden(
                raster_path=raster_file,
                year=year,
                disease=disease,
                source=source,
                version=f"MAP_{year}",
                metric_type='incidence_rate',
                stat='mean'
            )
            results[year] = stats
            print(f"  ✓ Loaded {stats['loaded']} locations for {year}")
            
        except Exception as e:
            print(f"  ❌ Failed to load {year}: {e}")
            results[year] = {'error': str(e)}
    
    return results

    save_best_join = ee.Join.saveFirst(matchKey='spatial_match')
    
    # Define the link: Match your "Name" column to OSM's "name" column
    # Note: OSM names are often lowercase, so we use a fuzzy match or just equals
    filter_match = ee.Filter.equals(
        leftField='Name', 
        rightField='name'
    )

    # Execute Join
    joined_col = save_best_join.apply(my_list, osm_reference, filter_match)

    # 5. Extract Coordinates from the joined "spatial_match"
    def extract_joined_coords(feature):
        matched_feat = ee.Feature(feature.get('spatial_match'))
        # If no match was found, this would stay null, so we handle that:
        return feature.set({
            'lon': ee.Algorithms.If(matched_feat, matched_feat.geometry().coordinates().get(0), 0),
            'lat': ee.Algorithms.If(matched_feat, matched_feat.geometry().coordinates().get(1), 0)
        })

    # Map the extraction and filter out those we couldn't find (the 0s)
    facilities_final = joined_col.map(extract_joined_coords).filter(ee.Filter.neq('lat', 0))

    # 6. Select and Export
    facilities_export = facilities_final.select(
        ['Name', 'Facility type', 'Owner', 'County', 'lat', 'lon']
    )

    count = facilities_export.size().getInfo()
    print(f"Matched {count} facilities with coordinates in {search_name_upper}.")

    if count > 0:
        task = ee.batch.Export.table.toDrive(
            collection=facilities_export,
            description=f'{county_name}_geocoded_facilities',
            folder='GEE_exports',
            fileFormat='CSV'
        )
        task.start()
        print("Geocoding export started.")
