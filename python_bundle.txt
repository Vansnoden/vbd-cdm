def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

def clip_era5_by_shapefile(grib_file_path, shapefile_path, filter_params=None, 
                          output_dir='./clipped_data', buffer_degrees=0.5):
    """
    Clip ERA5-Land GRIB files to features from any shapefile.
    (parameters unchanged)
    """
    os.makedirs(output_dir, exist_ok=True)

    def load_shapefile_features():
        """Load and filter features from shapefile (uses closure for filter_params, shapefile_path)"""
        try:
            gdf = gpd.read_file(shapefile_path)
            if gdf.crs is None:
                gdf = gdf.set_crs('EPSG:4326', allow_override=True)
            print(f"Shapefile loaded: {len(gdf)} features")
            print(f"Available columns: {list(gdf.columns)}")

            if filter_params is None:
                return gdf

            if isinstance(filter_params, dict):
                column = filter_params.get('column')
                value = filter_params.get('value')
                if column not in gdf.columns:
                    raise ValueError(f"Column '{column}' not found in shapefile")
                filtered_gdf = gdf[gdf[column] == value]

            elif isinstance(filter_params, list):
                masks = []
                for param in filter_params:
                    column = param.get('column')
                    value = param.get('value')
                    if column not in gdf.columns:
                        print(f"Warning: Column '{column}' not found, skipping")
                        continue
                    masks.append(gdf[column] == value)
                if not masks:
                    raise ValueError("No valid filter conditions found")
                combined_mask = masks[0]
                for mask in masks[1:]:
                    combined_mask = combined_mask | mask
                filtered_gdf = gdf[combined_mask]
            else:
                raise ValueError("filter_params must be dict, list of dicts, or None")

            print(f"Filtered to {len(filtered_gdf)} feature(s)")
            return filtered_gdf
        except Exception as e:
            print(f"Error loading shapefile: {e}")
            raise

    
    def process_feature(feature_gdf, feature_name, grib_file_path, output_dir):
        print(f"\nProcessing {feature_name}...")
        bbox = feature_gdf.total_bounds
        lon_slice = slice(bbox[0] - buffer_degrees, bbox[2] + buffer_degrees)
        lat_slice = slice(bbox[3] + buffer_degrees, bbox[1] - buffer_degrees)

        try:
            # Open all variable groups lazily with Dask chunks
            print("Opening GRIB file with lazy loading...")
            ds_list = cfgrib.open_datasets(
                grib_file_path,
                chunks={'time': 100},          # Load time in chunks of 100 steps
                backend_kwargs={'errors': 'raise'}
            )
            print(f"Found {len(ds_list)} separate variable groups.")

            # Subset each dataset to the bounding box and drop auxiliary variables
            subsets = []
            for i, ds in enumerate(ds_list):
                print(f"  Processing group {i+1}: {list(ds.data_vars)}")
                # Determine dimension names (CF convention)
                lon_dim = 'longitude' if 'longitude' in ds.dims else 'lon'
                lat_dim = 'latitude' if 'latitude' in ds.dims else 'lat'
                # Subset – this returns a lazy view (no data loaded yet)
                ds_sel = ds.sel(**{lon_dim: lon_slice, lat_dim: lat_slice})
                
                # Drop variables that often cause merge conflicts (non‑data coordinates)
                vars_to_drop = ['step', 'valid_time', 'number', 'surface', 'isobaricInhPa']
                to_drop = [v for v in vars_to_drop if v in ds_sel.variables]
                if to_drop:
                    ds_sel = ds_sel.drop_vars(to_drop)
                
                subsets.append(ds_sel)

            if not subsets:
                raise ValueError("No datasets loaded from GRIB file.")

            # Merge all subsets lazily – still no data loaded
            print("Merging variable groups (lazy)...")
            ds = xr.merge(subsets, compat='override', join='outer')
            print(f"Merged dataset variables: {list(ds.data_vars)}")

        except Exception as e:
            print(f"Error loading/processing GRIB: {e}")
            traceback.print_exc()
            return None

        # Ensure CRS for clipping (lazy, no data loaded)
        try:
            if not hasattr(ds, 'rio') or not ds.rio.crs:
                ds = ds.rio.write_crs("EPSG:4326")
        except Exception as e:
            print("Error setting CRS:")
            traceback.print_exc()
            return None

        # Clip to feature boundary (lazy)
        try:
            print("Clipping to feature boundary (lazy)...")
            clipped = ds.rio.clip(
                feature_gdf.geometry.apply(mapping),
                feature_gdf.crs,
                all_touched=True,
                drop=True
            )
        except Exception as e:
            print("Error during clipping:")
            traceback.print_exc()
            return None

        # Generate output filename
        base_name = os.path.basename(grib_file_path).replace('.grib', '').replace('.grb', '')
        safe_feature_name = feature_name.replace(' ', '_').replace('/', '_')
        output_path = os.path.join(output_dir, f"{base_name}_{safe_feature_name}.nc")

        # Save with compression – this triggers the actual computation chunk by chunk
        try:
            print(f"Saving to {output_path} (this will trigger computation)...")
            encoding = {var: {'zlib': True, 'complevel': 1, 'dtype': 'float32'}
                        for var in clipped.data_vars}
            # Use compute=False to write lazily, but we want to actually write now.
            # to_netcdf with compute=True will compute and write.
            clipped.to_netcdf(output_path, encoding=encoding, compute=True)
        except Exception as e:
            print("Error saving file:")
            traceback.print_exc()
            return None

        print(f"Success! Clipped shape: {clipped[list(clipped.data_vars)[0]].shape}")
        return output_path


    # --- Main flow of clip_era5_by_shapefile ---
    filtered_gdf = load_shapefile_features()
    if filtered_gdf.empty:
        print("No features found after filtering")
        return []

    output_files = []
    for idx, row in filtered_gdf.iterrows():
        single_feature_gdf = gpd.GeoDataFrame([row], geometry='geometry', crs=filtered_gdf.crs)

        # Try to create a meaningful feature name
        feature_name = None
        for name_col in ['NAME', 'name', 'COUNTY', 'county', 'ADM2_NAME', 'ADM1_NAME']:
            if name_col in row.index and pd.notna(row[name_col]):
                feature_name = str(row[name_col])
                break
        if feature_name is None:
            feature_name = f"feature_{idx}"
        for id_col in ['ID', 'id', 'OBJECTID', 'FID']:
            if id_col in row.index and pd.notna(row[id_col]):
                feature_name = f"{feature_name}_{row[id_col]}"
                break

        output_file = process_feature(single_feature_gdf, feature_name, grib_file_path, output_dir)
        if output_file:
            output_files.append(output_file)

    return output_files    


def clip_era5_to_kenya_counties(grib_file_path, county_names, shapefile_path=None, 
                               county_column='COUNTY', output_dir='./clipped_data'):
    """
    Convenience function specifically for Kenya counties.
    
    Parameters:
    -----------
    grib_file_path : str
        Path to the input GRIB file
    county_names : list or str
        Name(s) of Kenyan county/counties to clip to
    shapefile_path : str, optional
        Path to Kenya counties shapefile. If None, tries default locations
    county_column : str
        Column name in shapefile containing county names
    output_dir : str
        Directory to save clipped NetCDF files
    
    Returns:
    --------
    list
        Paths to saved clipped files
    """
    
    # Convert county_names to list if string
    if isinstance(county_names, str):
        county_names = [county_names]
    
    # Try to find Kenya shapefile if not provided
    if shapefile_path is None:
        possible_paths = [
            "kenya_counties.shp",
            "/path/to/kenya/counties.shp",  # Add your common paths here
            "data/shapefiles/kenya/admin2.shp"
        ]
        
        for path in possible_paths:
            if os.path.exists(path):
                shapefile_path = path
                print(f"Using shapefile: {shapefile_path}")
                break
        
        if shapefile_path is None:
            raise FileNotFoundError(
                "Kenya shapefile not found. Please provide shapefile_path parameter or "
                "download from: https://data.humdata.org/dataset/ken-administrative-boundaries"
            )
    
    # Build filter parameters
    if len(county_names) == 1:
        filter_params = {'column': county_column, 'value': county_names[0]}
    else:
        filter_params = [
            {'column': county_column, 'value': county_name}
            for county_name in county_names
        ]
    
    # Use the general function
    return clip_era5_by_shapefile(
        grib_file_path=grib_file_path,
        shapefile_path=shapefile_path,
        filter_params=filter_params,
        output_dir=output_dir
    )


def clip_era5_to_region(grib_file_path, geometry, region_name, output_dir='./clipped_data'):
    """
    Clip ERA5 data to a custom geometry (GeoDataFrame, GeoSeries, or Shapely geometry).
    
    Parameters:
    -----------
    grib_file_path : str
        Path to the input GRIB file
    geometry : geopandas.GeoDataFrame, geopandas.GeoSeries, or shapely.geometry
        Geometry to clip to
    region_name : str
        Name for the output file
    output_dir : str
        Directory to save clipped NetCDF file
    
    Returns:
    --------
    str
        Path to saved clipped file
    """
    
    import pandas as pd
    
    os.makedirs(output_dir, exist_ok=True)
    
    # Convert geometry to GeoDataFrame if needed
    if isinstance(geometry, gpd.GeoDataFrame):
        gdf = geometry
    elif isinstance(geometry, gpd.GeoSeries):
        gdf = gpd.GeoDataFrame(geometry=geometry)
    else:
        # Assume it's a shapely geometry
        gdf = gpd.GeoDataFrame(geometry=[geometry], crs='EPSG:4326')
    
    # Ensure CRS is set
    if gdf.crs is None:
        gdf = gdf.set_crs('EPSG:4326', allow_override=True)
    
    # Create a temporary shapefile
    with tempfile.TemporaryDirectory() as tmpdir:
        temp_shapefile = os.path.join(tmpdir, 'temp_region.shp')
        gdf.to_file(temp_shapefile)
        
        # Use the general function with no filtering
        output_files = clip_era5_by_shapefile(
            grib_file_path=grib_file_path,
            shapefile_path=temp_shapefile,
            filter_params=None,  # Use all features (just one in this case)
            output_dir=output_dir
        )
    
    # Rename the output file to use the region_name
    if output_files:
        original_path = output_files[0]
        base_name = os.path.basename(grib_file_path).replace('.grib', '').replace('.grb', '')
        new_path = os.path.join(output_dir, f"{base_name}_{region_name.replace(' ', '_')}.nc")
        
        # Rename the file
        if os.path.exists(original_path):
            os.rename(original_path, new_path)
            return new_path
    
    return None


def batch_clip_era5(grib_files, shapefile_path, filter_params=None, output_base_dir='./clipped_data'):
    """
    Batch clip multiple GRIB files using the same shapefile and filters.
    
    Parameters:
    -----------
    grib_files : list of str
        Paths to GRIB files
    shapefile_path : str
        Path to the shapefile
    filter_params : dict or list of dicts, optional
        Filter parameters (same as clip_era5_by_shapefile)
    output_base_dir : str
        Base directory for output
    
    Returns:
    --------
    dict
        Dictionary mapping input files to lists of output files
    """
    
    results = {}
    
    for grib_file in grib_files:
        print(f"\nProcessing: {grib_file}")
        
        # Create output subdirectory based on input filename
        base_name = os.path.basename(grib_file).replace('.grib', '').replace('.grb', '')
        output_dir = os.path.join(output_base_dir, base_name)
        
        try:
            output_files = clip_era5_by_shapefile(
                grib_file_path=grib_file,
                shapefile_path=shapefile_path,
                filter_params=filter_params,
                output_dir=output_dir
            )
            
            results[grib_file] = output_files
            print(f"Generated {len(output_files)} output files")
            
        except Exception as e:
            print(f"Error processing {grib_file}: {e}")
            results[grib_file] = []
            traceback.print_exc()
    
    return results


def ensure_dates_exist(start_date, end_date):
    session = SessionLocal()

            exists = session.query(DimTime).filter_by(date_id=date_id).first()
            if not exists:
                dt = DimTime(
                    date_id=date_id,
                    date=date,
                    year=date.year,
                    month=date.month,
                    week=date.isocalendar()[1],
                    day_of_year=date.timetuple().tm_yday,
                    season='dry' if date.month in [11,12,1,2,3] else 'wet',  # customize for your region
                    is_holiday='no'
                )
                session.add(dt)
        session.commit()
    finally:
        session.close()

def open_grib(filepath, variable=None):
    """Open a GRIB file with xarray + cfgrib engine."""
    # If variable is specified, try to filter at open time (if file contains multiple)
    kwargs = {'engine': 'cfgrib'}
    if variable:
        kwargs['filter_by_keys'] = {'shortName': variable}
    return xr.open_dataset(filepath, **kwargs)



def extract_point(ds, lat, lon, variable):
    """Extract value at the nearest grid point to (lat, lon)."""
    # Handle longitude wrapping if needed (assume ds uses 0-360 or -180-180)
    # Adjust lon to match dataset's coordinate system
    if hasattr(ds, 'longitude'):
        lon_vals = ds.longitude.values
        if lon_vals.min() >= 0 and lon < 0:
            lon += 360
        elif lon_vals.max() <= 180 and lon > 180:
            lon -= 360
    point = ds[variable].sel(latitude=lat, longitude=lon, method='nearest')
    return point.values.item() if point.values.size == 1 else float(point.values.mean())


def extract_polygon(ds, polygon, variable):
    """
    Average variable over a polygon (shapely Polygon) using rioxarray.
    Returns mean value (float) or None if no overlap.
    """
    # Ensure dataset has a proper CRS (usually WGS84)
    ds = ds.rio.write_crs("EPSG:4326")
    # Clip to polygon
    try:
        clipped = ds[variable].rio.clip([polygon], crs="EPSG:4326")
        # Compute mean over all remaining grid cells
        mean_val = clipped.mean(dim=['latitude', 'longitude']).values
        return mean_val.item() if mean_val.size == 1 else float(mean_val)
    except Exception:
        # No overlap or error
        return None


def aggregate_to_daily(ds, variable, method='mean'):
    """Resample sub-daily data to daily using method (mean, sum, min, max)."""
    # Ensure time is a datetime coordinate
    if 'time' not in ds.dims:
        return ds  # already daily?
    daily = ds[variable].resample(time='1D').reduce(method)
    return daily

class DatabaseLoader:
    def __init__(self, db_session: Session):
        self.session = db_session
    

    def bulk_load_cases(self, cases_data: List[Dict]):
        """Load multiple cases efficiently"""
        for case_dict in cases_data:
            # Validate with Pydantic
            case = schemas.HostCaseCreate(**case_dict)
            
            # Convert to ORM model
            db_case = models.HostCase(**case.dict())
            
            self.session.add(db_case)
        
        self.session.commit()
    

    def upsert_location(self, location_data: Dict):
        """Insert or update location"""
        location = self.session.query(models.Location)\
            .filter(models.Location.location_code == location_data['location_code'])\
            .first()
        
        if location:
            # Update existing
            for key, value in location_data.items():
                setattr(location, key, value)
        else:
            # Create new
            location = models.Location(**location_data)
            self.session.add(location)
        
        self.session.commit()
        return location
    
    
    def load_from_dataframe(self, df: pd.DataFrame, model_class):
        """Generic method to load pandas DataFrame to any model"""
        records = df.to_dict('records')
        for record in records:
            # Handle missing columns
            filtered_record = {k: v for k, v in record.items() 
                             if hasattr(model_class, k)}
            db_obj = model_class(**filtered_record)
            self.session.add(db_obj)
        
        self.session.commit()


    def load_from_grib(self, start_date, end_date, grib_file_pattern):
        """
        grib_file_pattern: string with {date} placeholder, e.g., '/data/era5_{date}.grib'
        """
        # 1. Load all locations with geometry
        locations = self._get_locations_as_geodataframe()
        
        # 2. Generate date range
        date_range = pd.date_range(start_date, end_date)
        
        for date in date_range:
            date_str = date.strftime('%Y-%m-%d')
            date_id = date_str
            file_path = grib_file_pattern.format(date=date.strftime('%Y%m%d'))
            
            # Dictionary to hold combined data for this date: key = (loc_code, date_id)
            records = {}
            for _, row in locations.iterrows():
                records[(row.location_code, date_id)] = {
                    'location_code': row.location_code,
                    'date_id': date_id
                }
            
            # Process each variable from mapping
            for grib_var, mapping in mappings.GRIB_VARIABLE_MAP.items():
                try:
                    ds = grib_utils.open_grib(file_path, variable=grib_var)
                    
                    # If sub-daily, aggregate to daily
                    if 'time' in ds.dims and ds.dims['time'] > 1:
                        ds = grib_utils.aggregate_to_daily(ds, grib_var)
                    
                    # Extract for each location
                    for _, row in locations.iterrows():
                        geom = row.geometry
                        if geom is None or geom.is_empty:
                            # Use centroid
                            lat, lon = row.centroid_lat, row.centroid_lon
                            if lat is None or lon is None:
                                continue
                            val = grib_utils.extract_point(ds, lat, lon, grib_var)
                        else:
                            val = grib_utils.extract_polygon(ds, geom, grib_var)
                        
                        if val is None:
                            continue
                        
                        # Apply unit conversion
                        if mapping['convert']:
                            val = mapping['convert'](val)
                        
                        # Store in records
                        key = (row.location_code, date_id)
                        records[key][mapping['column']] = val
                        
                except Exception as e:
                    print(f"Error processing variable {grib_var} for {file_path}: {e}")
                    continue
            
            # Now insert records into database
            for key, data in records.items():
                # Skip if no environmental data was collected (all vars missing)
                if any(col in data for col in ['temperature_mean', 'rainfall_total', 
                                                'temperature_min', 'temperature_max',
                                                'relative_humidity']):
                    # Check if record already exists
                    existing = self.session.query(EnvironmentalFactor).filter_by(
                        location_code=data['location_code'],
                        date_id=data['date_id']
                    ).first()
                    if existing:
                        # Update existing record with new values (optional)
                        for col, val in data.items():
                            if col not in ['location_code', 'date_id']:
                                setattr(existing, col, val)
                    else:
                        env_factor = EnvironmentalFactor(**data)
                        self.session.add(env_factor)
            
            self.session.commit()
            print(f"Loaded environmental data for {date_str}")


    def _get_locations_as_geodataframe(self):
        """Load all locations with WKT geometry into a GeoDataFrame."""
        locations = self.session.query(Location).all()
        data = []
        for loc in locations:
            geom = wkt.loads(loc.geometry_wkt) if loc.geometry_wkt else None
            data.append({
                'location_code': loc.location_code,
                'geometry': geom,
                'centroid_lat': loc.centroid_lat,
                'centroid_lon': loc.centroid_lon
            })
        # Create GeoDataFrame and explicitly set the geometry column
        gdf = gpd.GeoDataFrame(data, crs='EPSG:4326', geometry='geometry')
        return gdf

    
    def load_from_netcdf(self, start_date, end_date, file_path_func):
        """
        file_path_func: a function with signature (location_code, date) -> str
                       that returns the full path to the NetCDF file.
        Example: lambda loc, d: f"/data/{loc}_{d.strftime('%Y%m%d')}.nc"
        """
        # 1. Load all locations with geometries
        locations = self._get_locations_as_geodataframe()
        if locations.empty:
            print("No locations found in database. Run load_locations.py first.")
            return

        # 2. Generate date range
        date_range = pd.date_range(start_date, end_date)

        # 3. For each location and date, extract and store
        for date in date_range:
            date_str = date.strftime('%Y-%m-%d')
            date_id = date_str

            for _, loc_row in locations.iterrows():
                loc_code = loc_row.location_code
                file_path = file_path_func(loc_code, date)

                if not Path(file_path).exists():
                    print(f"File not found: {file_path}")
                    continue

                # Dictionary to hold data for this location+date
                record = {'location_code': loc_code, 'date_id': date_id}

                # Open the NetCDF file once per location+date
                try:
                    ds = netcdf_utils.open_netcdf(file_path)
                except Exception as e:
                    print(f"Error opening {file_path}: {e}")
                    continue

                # Process each variable in the mapping that exists in the dataset
                for nc_var, mapping in mappings.NC_VARIABLE_MAP.items():
                    if nc_var not in ds:
                        # Variable not in this file – skip silently
                        continue

                    try:
                        # If sub-daily, aggregate to daily using the specified method
                        if 'time' in ds.dims and ds.sizes['time'] > 1:
                            da = netcdf_utils.aggregate_to_daily(
                                ds, nc_var, method=mapping.get('agg_method', 'mean')
                            )
                        else:
                            da = ds[nc_var]

                        # Extract value for this location (using polygon or centroid)
                        geom = loc_row.geometry
                        if geom is None or geom.is_empty:
                            lat, lon = loc_row.centroid_lat, loc_row.centroid_lon
                            if lat is None or lon is None:
                                continue
                            val = netcdf_utils.extract_point(da, lat, lon)
                        else:
                            val = netcdf_utils.extract_polygon(da, geom)

                        if val is None:
                            continue

                        # Apply unit conversion
                        if mapping['convert']:
                            val = mapping['convert'](val)

                        record[mapping['column']] = val

                    except Exception as e:
                        print(f"Error processing {nc_var} for {loc_code} on {date_str}: {e}")
                        continue

                # After processing all variables, upsert the record
                if len(record) > 2:  # at least one environmental variable besides loc/date
                    self._upsert_environmental(record)

            # Commit after each date to avoid large transactions
            self.session.commit()
            print(f"Loaded environmental data for {date_str}")


    def _get_locations_as_geodataframe(self):

        :param schema_class: Optional Pydantic schema for validation (e.g., HostCaseCreate)
        :param column_mapping: Dict mapping file columns to model attributes

        if column_mapping:

        records = df.to_dict('records')

            validated_records = []
            for rec in records:
                # Filter to only keys present in schema (optional)

                filtered = {k: v for k, v in rec.items() if k in schema_fields}
                validated = schema_class(**filtered)
                validated_records.append(validated.model_dump())
            records = validated_records
        
        # Use existing bulk load method (or generic load_from_dataframe)
        self.load_from_dataframe(pd.DataFrame(records), model_class)

def open_netcdf(filepath):
    """Open a NetCDF file with xarray."""

def extract_point(da, lat, lon):
    """
    Extract value from a DataArray at the nearest grid point.
    da: DataArray with dimensions latitude, longitude (and possibly time).
    """
    # Align longitude system if needed
    if 'longitude' in da.dims:
        lon_vals = da.longitude.values
        if lon_vals.min() >= 0 and lon < 0:
            lon += 360
        elif lon_vals.max() <= 180 and lon > 180:
            lon -= 360
    # Select nearest point – if time dimension exists, take first time?
    # For daily aggregated data, time should be a single step.
    point = da.sel(latitude=lat, longitude=lon, method='nearest')
    # If time still present, average over time (should be single after aggregation)
    if 'time' in point.dims and point.sizes['time'] > 1:
        point = point.mean(dim='time')
    return point.values.item() if point.values.size == 1 else float(point.values.mean())


def extract_polygon(da, polygon):
    """
    Average variable over a polygon using rioxarray.
    da: DataArray with latitude/longitude dimensions.
    """
    # Ensure CRS
    da = da.rio.write_crs("EPSG:4326")
    try:
        clipped = da.rio.clip([polygon], crs="EPSG:4326")
        mean_val = clipped.mean(dim=['latitude', 'longitude']).values
        # If time still present, average over time
        if 'time' in clipped.dims:
            mean_val = mean_val.mean()
        return mean_val.item() if mean_val.size == 1 else float(mean_val)
    except Exception:
        return None


def aggregate_to_daily(ds, variable, method='mean'):
    """
    Resample sub-daily data to daily using method.
    Returns a DataArray with a single time step per day.
    """
    # Ensure time is datetime
    if not np.issubdtype(ds.time.dtype, np.datetime64):
        ds['time'] = pd.to_datetime(ds.time.values)
    daily = ds[variable].resample(time='1D').reduce(method)
    return daily

def parse_filename(filepath):
    """
    Extract year and location from filename.
    Expected pattern: {year}_data_{location}_{id}.nc
    Example: 2024_data_Mombasa_141.nc
    Returns (year, location) or (None, None) if no match.
    """
    pattern = r'(?P<year>\d{4})_data_(?P<location>[A-Za-z]+)_\d+\.nc$'
    match = re.search(pattern, filepath.name)
    if not match:
        return None, None
    year = int(match.group('year'))

def load_annual_netcdf_files(data_root):
    """
    Process all annual NetCDF files in data_root and load into environmental_factor.
    """
    # Get database session
    db = SessionLocal()

    # Load all locations with their geometries and centroids
    locations = db.query(Location).all()
    loc_dict = {}
    for loc in locations:
        geom = wkt.loads(loc.geometry_wkt) if loc.geometry_wkt else None
        loc_dict[loc.location_code] = {
            'geometry': geom,
            'centroid_lat': loc.centroid_lat,
            'centroid_lon': loc.centroid_lon
        }

    if not loc_dict:
        print("No locations found in database. Run load_location_data.py first.")
        db.close()
        return

    # Find all NetCDF files recursively
    nc_files = list(Path(data_root).rglob("*.nc"))
    print(f"Found {len(nc_files)} NetCDF files.")

    for nc_file in nc_files:
        year, loc_code = parse_filename(nc_file)
        if loc_code is None:
            print(f"Skipping unrecognized file: {nc_file.name}")
            continue
        if loc_code not in loc_dict:
            print(f"Location '{loc_code}' not found in database. Skipping {nc_file.name}")
            continue

        print(f"\nProcessing {nc_file.name} (location: {loc_code}, year: {year})")

        # Open the dataset
        try:
            ds = xr.open_dataset(nc_file)
        except Exception as e:
            print(f"  Error opening file: {e}")
            continue

        # Get all unique dates from the time coordinate
        times = pd.to_datetime(ds.time.values)
        dates = sorted(set(t.date() for t in times))
        if not dates:
            print("  No time data found.")
            ds.close()
            continue

        # Ensure all dates exist in dim_time
        min_date = dates[0].strftime('%Y-%m-%d')
        max_date = dates[-1].strftime('%Y-%m-%d')
        ensure_dates_exist(min_date, max_date)

        # Get geometry info for this location
        geom_info = loc_dict[loc_code]
        geometry = geom_info['geometry']
        lat, lon = geom_info['centroid_lat'], geom_info['centroid_lon']

        # Process each variable defined in the mapping

        for nc_var, mapping in NC_VARIABLE_MAP.items():
            if nc_var not in ds:
                continue

            print(f"  Processing variable: {nc_var}")

            # Get aggregation method name from mapping (default 'mean')
            method_name = mapping.get('agg_method', 'mean')
            
            # Convert method name to a callable
            method_func = {
                'mean': np.mean,
                'sum': np.sum,
                'max': np.max,
                'min': np.min,
            }.get(method_name)
            
            if method_func is None:
                print(f"    Warning: Unknown method '{method_name}', using np.mean")
                method_func = np.mean

            # Aggregate sub-daily data to daily using the callable
            try:
                daily_da = netcdf_utils.aggregate_to_daily(
                    ds, nc_var, method=method_func
                )
            except Exception as e:
                print(f"    Error aggregating: {e}")
                continue

            # Now iterate over each day
            for time_idx in range(len(daily_da.time)):
                day_slice = daily_da.isel(time=time_idx)
                current_date = pd.to_datetime(daily_da.time.values[time_idx]).date()
                date_id = current_date.strftime('%Y-%m-%d')

                # Extract value for this location
                try:
                    if geometry is None or geometry.is_empty:
                        if lat is None or lon is None:
                            continue
                        value = netcdf_utils.extract_point(day_slice, lat, lon)
                    else:
                        value = netcdf_utils.extract_polygon(day_slice, geometry)

                    if value is None:
                        continue

                    # Apply unit conversion
                    if mapping.get('convert'):
                        value = mapping['convert'](value)

                    # Prepare record for upsert
                    record = {
                        'location_code': loc_code,
                        'date_id': date_id,
                        mapping['column']: value
                    }

                    # Upsert into database
                    existing = db.query(EnvironmentalFactor).filter_by(
                        location_code=loc_code,
                        date_id=date_id
                    ).first()

                    if existing:
                        setattr(existing, mapping['column'], value)
                    else:
                        # Create full record with all columns (others will be NULL)
                        full_record = {
                            'location_code': loc_code,
                            'date_id': date_id,
                            mapping['column']: value
                        }
                        env_factor = EnvironmentalFactor(**full_record)
                        db.add(env_factor)

                except Exception as e:
                    print(f"    Error extracting value for {date_id}: {e}")
                    continue

        # Commit after each file to avoid large transactions
        db.commit()
        ds.close()
        print(f"  Finished processing {nc_file.name}")

    db.close()
    print("\nAll files processed.")

    """
    Load environmental data from NetCDF files for all locations.
    :param start_date: string 'YYYY-MM-DD'
    :param end_date: string 'YYYY-MM-DD'
    :param data_dir: base directory containing the NetCDF files
    :param file_pattern_func: optional function(location_code, date) -> full file path.
                              If None, a default pattern is used:
                              f"{data_dir}/{location_code}_{date.strftime('%Y%m%d')}.nc"
    """
    session = SessionLocal()
    try:
        # 1. Get all locations with geometries
        locations = session.query(Location).all()
        if not locations:
            print("No locations found. Run load_location_data first.")
            return

        # Convert to a list of dicts with geometry
        loc_list = []
        for loc in locations:
            geom = wkt.loads(loc.geometry_wkt) if loc.geometry_wkt else None
            loc_list.append({
                'code': loc.location_code,
                'geometry': geom,
                'centroid_lat': loc.centroid_lat,
                'centroid_lon': loc.centroid_lon
            })

        # 2. Define default file pattern if not provided
        if file_pattern_func is None:
            def default_pattern(loc_code, date):
                return Path(data_dir) / f"{loc_code}_{date.strftime('%Y%m%d')}.nc"
            file_pattern_func = default_pattern

        # 3. Generate date range
        date_range = pd.date_range(start_date, end_date)

        # 4. For each date and location, process t2m
        for date in date_range:
            date_str = date.strftime('%Y-%m-%d')
            date_id = date_str

            for loc in loc_list:
                loc_code = loc['code']
                file_path = file_pattern_func(loc_code, date)

                if not Path(file_path).exists():
                    # Silently skip missing files (or print debug)
                    continue

                try:
                    ds = open_netcdf(file_path)
                except Exception as e:
                    print(f"Error opening {file_path}: {e}")
                    continue

                # We only have t2m in the mapping (others may be added later)
                # Hardcode t2m for now – you can move this to a mapping module later
                if 't2m' not in ds:
                    print(f"Variable 't2m' not found in {file_path}")
                    continue

                # Aggregate to daily if needed
                if 'time' in ds.dims and ds.sizes['time'] > 1:
                    da = aggregate_to_daily(ds, 't2m', method='mean')

                    date_id=date_id

                if existing:
                    existing.temperature_mean = val_celsius
                else:
                    env = EnvironmentalFactor(
                        location_code=loc_code,

                    )

    if gdf.crs and gdf.crs.to_string() != 'EPSG:4326':
        gdf = gdf.to_crs('EPSG:4326')
    
    session = SessionLocal()
    try:
        for _, row in gdf.iterrows():
            # Try to get a name from common attribute fields
            # Adjust these field names to match your actual data
            name = None
            for col in ['NAME', 'name', 'COUNTY', 'county', 'ADMIN', 'admin', 'adm1_name']:
                if col in row and row[col]:
                    name = str(row[col])
                    break
            if not name:
                name = f"Feature_{_}"  # fallback
            
            # Generate a unique location code (e.g., lowercase with underscores)
            location_code = name.lower().replace(' ', '_').replace('-', '_')
            
            # Get geometry as WKT
            geom_wkt = row.geometry.wkt if row.geometry else None
            
            # Compute centroid
            if row.geometry:
                centroid = row.geometry.centroid
                centroid_lat = centroid.y
                centroid_lon = centroid.x
            else:
                centroid_lat = centroid_lon = None
            
            existing = session.query(Location).filter_by(location_code=location_code).first()
            if existing:
                # Update existing record
                existing.name = name
                existing.admin_level = admin_level
                existing.parent_location_code = parent_code
                existing.geometry_wkt = geom_wkt
                existing.centroid_lat = centroid_lat
                existing.centroid_lon = centroid_lon
                print(f"Updated: {location_code}")
            else:
                # Insert new record
                location = Location(
                    location_code = location_code,
                    name = name,
                    admin_level = admin_level,
                    parent_location_code = parent_code,
                    geometry_wkt = geom_wkt,
                    centroid_lat = centroid_lat,
                    centroid_lon = centroid_lon
                )
                session.add(location)
                print(f"Inserted: {location_code}")

        session.commit()
        print(f"Loaded {len(gdf)} locations from {gpkg_path}")
    except Exception as e:
        session.rollback()
        print(f"Error loading GeoPackage: {e}")
    finally:
        session.close()

# load_vector_data.py – Generic loader using column mapping

import sys

def generate_location_code(lat, lon):
    """Generate a unique location code from coordinates."""
    # Replace negative sign with 'n' and decimal point with '_' for readability
    return f"point_{lat:.4f}_{lon:.4f}".replace('-', 'n').replace('.', '_')


def load_vector_files(data_dir, pattern="*.csv", column_mapping=None, **pandas_kwargs):
    """
    Load vector observation files using a column mapping dictionary.

    The column_mapping must contain special keys:
        __latitude__  : name of column with latitude
        __longitude__ : name of column with longitude
        __year__      : name of column with year
        __month__     : name of column with month
        __count__     : either a single column name or a list of column names to sum
    Additional keys (not starting with '__') are mapped directly to VectorObservation fields.
    """
    data_dir = Path(data_dir)
    files = list(data_dir.glob(pattern))
    if not files:
        print(f"No files matching {pattern} found in {data_dir}")
        return

    db = SessionLocal()
    try:
        loader = DatabaseLoader(db)

        for file_path in files:
            print(f"\nProcessing {file_path.name}...")

            # --- 1. Read file ---
            if file_path.suffix.lower() == '.csv':
                df = pd.read_csv(file_path, **pandas_kwargs)
            elif file_path.suffix.lower() in ['.xls', '.xlsx']:
                df = pd.read_excel(file_path, **pandas_kwargs)
            else:
                print(f"  Skipping unsupported file type: {file_path.suffix}")
                continue

            if df.empty:
                print("  File is empty.")
                continue

            # --- 2. Extract required columns from mapping ---
            lat_col = column_mapping.get('__latitude__')
            lon_col = column_mapping.get('__longitude__')
            year_col = column_mapping.get('__year__')
            month_col = column_mapping.get('__month__')
            count_spec = column_mapping.get('__count__')

            if not all([lat_col, lon_col, year_col, month_col, count_spec]):
                print("  Missing one or more required mapping keys (__latitude__, __longitude__, __year__, __month__, __count__). Skipping file.")
                continue

            # Check that required columns exist
            required_cols = [lat_col, lon_col, year_col, month_col]
            missing = [c for c in required_cols if c not in df.columns]
            if missing:
                print(f"  Missing columns: {missing}. Skipping file.")
                continue

            # --- 3. Handle count (single column or list of columns) ---
            if isinstance(count_spec, str):
                if count_spec not in df.columns:
                    print(f"  Count column '{count_spec}' not found. Using default count=1 for all rows.")
                    df['_count'] = 1
                else:
                    df['_count'] = pd.to_numeric(df[count_spec], errors='coerce').fillna(1)
            elif isinstance(count_spec, list):
                # For a list of columns, we sum them (but not typical for GBIF)
                missing_counts = [c for c in count_spec if c not in df.columns]
                if missing_counts:
                    print(f"  Missing count columns: {missing_counts}. Using default count=1.")
                    df['_count'] = 1
                else:
                    for c in count_spec:
                        df[c] = pd.to_numeric(df[c], errors='coerce').fillna(0)
                    df['_count'] = df[count_spec].sum(axis=1)
            else:
                print("  __count__ must be a string or a list. Skipping file.")
                continue

            # --- 4. Generate location codes and upsert locations ---
            df['_lat'] = df[lat_col].astype(float)
            df['_lon'] = df[lon_col].astype(float)
            df['location_code'] = df.apply(
                lambda r: generate_location_code(r['_lat'], r['_lon']), axis=1
            )

            # Drop duplicates on location_code (keep first occurrence)
            unique_locs = df[['location_code', '_lat', '_lon']].drop_duplicates(subset=['location_code'])

            # Upsert unique locations
            for _, row in unique_locs.iterrows():
                existing = db.query(models.Location).filter_by(location_code=row['location_code']).first()
                if not existing:
                    point = Point(row['_lon'], row['_lat'])
                    loc = models.Location(
                        location_code=row['location_code'],
                        name=f"Sampling site at {row['_lat']:.4f}, {row['_lon']:.4f}",
                        admin_level='site',
                        geometry_wkt=point.wkt,
                        centroid_lat=row['_lat'],
                        centroid_lon=row['_lon']
                    )
                    db.add(loc)
            db.commit()
            print(f"  Upserted {len(unique_locs)} location sites.")

            # --- 5. Construct date_id and ensure dim_time exists ---
            # Convert year and month to numeric, coercing errors to NaN
            df['year'] = pd.to_numeric(df[year_col], errors='coerce')
            df['month'] = pd.to_numeric(df[month_col], errors='coerce')
            
            # Drop rows where year or month is NaN
            df = df.dropna(subset=['year', 'month'])
            
            # Keep only realistic years (1900–2100) and months 1–12
            df = df[(df['year'] >= 1900) & (df['year'] <= 2100) & (df['month'].between(1, 12))]
            
            if df.empty:
                print("  No rows with valid year/month after cleaning. Skipping file.")
                continue
            
            # Convert to integer (now safe)
            df['year'] = df['year'].astype(int)
            df['month'] = df['month'].astype(int)
            
            # Create date_id (first day of month)
            df['date_id'] = df.apply(
                lambda r: f"{r['year']:04d}-{r['month']:02d}-01", axis=1
            )
            
            # Compute min and max date from valid rows
            min_date = df['date_id'].min()
            max_date = df['date_id'].max()
            
            # Ensure dim_time records exist for all dates in this range
            ensure_dates_exist(min_date, max_date)

            # --- 6. Build field mapping for direct attributes ---
            field_mapping = {}
            for key, value in column_mapping.items():
                if not key.startswith('__') and value in df.columns:
                    field_mapping[key] = value

            # --- 7. Create validated records ---
            obs_records = []
            for _, row in df.iterrows():
                record = {
                    'location_code': row['location_code'],
                    'date_id': row['date_id'],
                    'count': int(row['_count']),
                }
                # Add directly mapped fields
                for out_field, src_col in field_mapping.items():
                    val = row[src_col]
                    if pd.notna(val):
                        record[out_field] = val

                # Validate with Pydantic
                try:
                    validated = schemas.VectorObservationCreate(**record)
                    obs_records.append(validated.model_dump())
                except Exception as e:
                    print(f"  Validation error for record {record}: {e}")

            if obs_records:
                loader.load_from_dataframe(pd.DataFrame(obs_records), models.VectorObservation)
                print(f"  Loaded {len(obs_records)} vector observations.")
            else:
                print("  No valid observation records.")

    finally:
        db.close()

def add_coords(feature):
    """Add lat and lon properties derived from the feature's geometry."""
    geom = feature.geometry()
    coords = geom.coordinates()
    return feature.set({
        'lat': coords.get(1),
        'lon': coords.get(0)
    })


def download_health_facility_occurrence_data(county_name: str):
    # 1. Setup Names
    search_name_upper = county_name.upper()

    # 2. Load your Tabular Asset (No coordinates)
    my_list = ee.FeatureCollection("projects/gen-lang-client-0359288668/assets/kenya-health-facilities-2017_08_02") \
                .filter(ee.Filter.eq('County', search_name_upper))

    # 3. Load a Spatial Reference (OSM Health Facilities - which HAVE coordinates)
    # We use a public dataset to "find" the points for your names
    osm_reference = ee.FeatureCollection("projects/sat-io/open-datasets/HOT_OSM/ken_health_facilities_points")

    # 4. Define a Join
    # We match 'Name' from your list to 'name' in OSM
    save_best_join = ee.Join.saveFirst(matchKey='spatial_match')
    
    # Define the link: Match your "Name" column to OSM's "name" column
    # Note: OSM names are often lowercase, so we use a fuzzy match or just equals
    filter_match = ee.Filter.equals(
        leftField='Name', 
        rightField='name'
    )

    # Execute Join
    joined_col = save_best_join.apply(my_list, osm_reference, filter_match)

    # 5. Extract Coordinates from the joined "spatial_match"
    def extract_joined_coords(feature):
        matched_feat = ee.Feature(feature.get('spatial_match'))
        # If no match was found, this would stay null, so we handle that:
        return feature.set({
            'lon': ee.Algorithms.If(matched_feat, matched_feat.geometry().coordinates().get(0), 0),
            'lat': ee.Algorithms.If(matched_feat, matched_feat.geometry().coordinates().get(1), 0)
        })

    # Map the extraction and filter out those we couldn't find (the 0s)
    facilities_final = joined_col.map(extract_joined_coords).filter(ee.Filter.neq('lat', 0))

    # 6. Select and Export
    facilities_export = facilities_final.select(
        ['Name', 'Facility type', 'Owner', 'County', 'lat', 'lon']
    )

    count = facilities_export.size().getInfo()
    print(f"Matched {count} facilities with coordinates in {search_name_upper}.")

    if count > 0:
        task = ee.batch.Export.table.toDrive(
            collection=facilities_export,
            description=f'{county_name}_geocoded_facilities',
            folder='GEE_exports',
            fileFormat='CSV'
        )
        task.start()
        print("Geocoding export started.")
