Configuration

Code 
cfg_tbl <- tibble::tibble(
  Setting = c(
    "study_name", "location", "country",
    "location_ids", "start_year", "end_year",
    "disease_code", "include_estimated_cases",
    "env_variables", "species_filter", "max_lag_months", "db_path"
  ),
  Value = c(
    cfg$study_name %||% "(missing)",
    cfg$location    %||% "(missing)",
    cfg$country     %||% "(missing)",
    if (length(cfg$location_ids)) paste(cfg$location_ids, collapse = ", ")
      else "(name-based resolution)",
    as.character(cfg$start_year %||% NA),
    as.character(cfg$end_year   %||% NA),
    if (is.null(cfg$disease_code)) "(none)" else cfg$disease_code,
    as.character(isTRUE(cfg$include_estimated_cases)),
    paste(cfg$env_variables %||% character(), collapse = ", "),
    if (is.null(cfg$species_filter)) "(all)"
      else paste(cfg$species_filter, collapse = ", "),
    as.character(cfg$max_lag_months %||% NA),
    cfg$db_path %||% "(not set)"
  )
)

Species filter coverage

Code 
sp <- qa$species
if (!is.null(cfg$species_filter) && is.data.frame(sp) && nrow(sp) > 0) {
  present <- tolower(sp$species_name)
  tibble::tibble(
    filter_value = cfg$species_filter,
    matched      = vapply(cfg$species_filter,
                          \(v) any(grepl(tolower(v), present, fixed = TRUE)),
                          logical(1))
  ) |> gt::gt() |> gt::tab_header(title = "Species filter coverage")
} else {
  cat("_Species filter coverage not applicable._")
}

  print(
    ggplot(mc, aes(x = month_start, y = case_count)) +
      geom_line(color = "#1E40AF") +
      geom_point(color = "#1E40AF", size = 1) +
      labs(x = NULL, y = "Cases / month",

      theme_minimal()
  )
} else {
  cat("_No observed case rows present._ ",
      "This is expected if the CDM carries vector, environment, ",
